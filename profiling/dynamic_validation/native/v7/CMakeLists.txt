cmake_minimum_required(VERSION 3.22)
project(dynamic_h2d_primitive_v7 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "This diagnostic requires Windows QPC")
endif()
set(LLAMA_ROOT "" CACHE PATH "Existing pinned llama source")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML import libraries")
set(STEADY_GENERATED_DIR "" CACHE PATH "Existing generated Executor includes")
set(CUDA_ROOT "" CACHE PATH "Existing CUDA toolkit")
set(CUPTI_ROOT "" CACHE PATH "Header dependency of retained common.hpp; no CUPTI capture/link")
foreach(path "${LLAMA_ROOT}/ggml/src/ggml-cuda/ggml-cuda.cu" "${LLAMA_ROOT}/ggml/src/ggml-backend.cpp" "${STEADY_GENERATED_DIR}/state_response_generated.inc" "${CUDA_ROOT}/include/cuda_runtime_api.h" "${CUPTI_ROOT}/include/cupti.h")
 if(NOT EXISTS "${path}")
  message(FATAL_ERROR "Explicit dependency missing: ${path}")
 endif()
endforeach()
set(libraries)
foreach(component base cpu cuda)
 find_library(GGML_${component} NAMES ggml-${component} PATHS "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda" PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
 list(APPEND libraries "${GGML_${component}}")
endforeach()
find_library(CUDART_LIBRARY NAMES cudart PATHS "${CUDA_ROOT}/lib/x64" NO_DEFAULT_PATH REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/h2d_primitive.cpp" H2D_PRIMITIVE_SOURCE_SHA256)
file(SHA256 "${LLAMA_ROOT}/ggml/src/ggml-cuda/ggml-cuda.cu" H2D_PRIMITIVE_ORIGINAL_CUDA_SHA256)
file(SHA256 "${LLAMA_ROOT}/ggml/src/ggml-backend.cpp" H2D_PRIMITIVE_ORIGINAL_BACKEND_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/h2d_primitive.cpp" "${LLAMA_ROOT}/ggml/src/ggml-cuda/ggml-cuda.cu" "${LLAMA_ROOT}/ggml/src/ggml-backend.cpp")
add_executable(dynamic-h2d-primitive-v7 h2d_primitive.cpp)
target_compile_features(dynamic-h2d-primitive-v7 PRIVATE cxx_std_17)
target_compile_definitions(dynamic-h2d-primitive-v7 PRIVATE GGML_SHARED GGML_BACKEND_SHARED HIGHLEVEL_CUDA=1 H2D_PRIMITIVE_SOURCE_SHA256="${H2D_PRIMITIVE_SOURCE_SHA256}" H2D_PRIMITIVE_ORIGINAL_CUDA_SHA256="${H2D_PRIMITIVE_ORIGINAL_CUDA_SHA256}" H2D_PRIMITIVE_ORIGINAL_BACKEND_SHA256="${H2D_PRIMITIVE_ORIGINAL_BACKEND_SHA256}")
target_include_directories(dynamic-h2d-primitive-v7 PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/ggml/src" "${LLAMA_ROOT}/vendor" "${CUDA_ROOT}/include" "${CUPTI_ROOT}/include" "${STEADY_GENERATED_DIR}")
target_link_libraries(dynamic-h2d-primitive-v7 PRIVATE ${libraries} "${CUDART_LIBRARY}" bcrypt)
if(MSVC)
 target_compile_options(dynamic-h2d-primitive-v7 PRIVATE /W4 /EHsc)
endif()
