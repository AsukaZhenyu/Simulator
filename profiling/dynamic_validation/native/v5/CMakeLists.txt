cmake_minimum_required(VERSION 3.22)
project(dynamic_external_v5 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "This scoped diagnostic requires Windows QPC")
endif()
set(LLAMA_ROOT "" CACHE PATH "Existing llama source")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML import libraries")
set(STEADY_GENERATED_DIR "" CACHE PATH "Bound generated Executor files")
set(CUDA_ROOT "" CACHE PATH "Existing CUDA toolkit")
set(CUPTI_ROOT "" CACHE PATH "Existing CUPTI toolkit")
foreach(path "${LLAMA_ROOT}/ggml/include/ggml.h" "${STEADY_GENERATED_DIR}/state_response_generated.inc" "${CUDA_ROOT}/include/cuda_runtime_api.h" "${CUPTI_ROOT}/include/cupti.h")
 if(NOT EXISTS "${path}")
  message(FATAL_ERROR "Explicit dependency missing: ${path}")
 endif()
endforeach()
set(libraries)
foreach(component base cpu cuda)
 find_library(GGML_${component} NAMES ggml-${component} PATHS "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda" PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
 list(APPEND libraries "${GGML_${component}}")
endforeach()
find_library(CUDART_LIBRARY NAMES cudart PATHS "${CUDA_ROOT}/lib/x64" NO_DEFAULT_PATH REQUIRED)
find_library(CUPTI_LIBRARY NAMES cupti PATHS "${CUPTI_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/external_probe.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/external_probe.cpp" EXTERNAL_V5_SOURCE_SHA256)
add_executable(dynamic-external-stages-v5 external_probe.cpp)
target_compile_features(dynamic-external-stages-v5 PRIVATE cxx_std_17)
target_compile_definitions(dynamic-external-stages-v5 PRIVATE GGML_SHARED GGML_BACKEND_SHARED HIGHLEVEL_CUDA=1 EXTERNAL_V5_SOURCE_SHA256="${EXTERNAL_V5_SOURCE_SHA256}")
target_include_directories(dynamic-external-stages-v5 PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/ggml/src" "${LLAMA_ROOT}/vendor" "${CUDA_ROOT}/include" "${CUPTI_ROOT}/include" "${STEADY_GENERATED_DIR}")
target_link_libraries(dynamic-external-stages-v5 PRIVATE ${libraries} "${CUDART_LIBRARY}" "${CUPTI_LIBRARY}" bcrypt psapi)
if(MSVC)
 target_compile_options(dynamic-external-stages-v5 PRIVATE /W4 /EHsc)
endif()
