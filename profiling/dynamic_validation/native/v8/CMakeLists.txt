cmake_minimum_required(VERSION 3.22)
project(dynamic_external_stage_v8 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "This diagnostic requires Windows QPC")
endif()
set(LLAMA_ROOT "" CACHE PATH "Existing pinned llama source")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML import libraries")
set(STEADY_GENERATED_DIR "" CACHE PATH "Existing generated Executor includes")
set(CUDA_ROOT "" CACHE PATH "Existing CUDA toolkit")
set(CUPTI_ROOT "" CACHE PATH "Existing CUPTI toolkit")
set(Python3_EXECUTABLE "" CACHE FILEPATH "Explicit existing Python for actor derivation check")
if(NOT EXISTS "${Python3_EXECUTABLE}")
 message(FATAL_ERROR "Explicit Python interpreter required")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" -m profiling.dynamic_validation.actor_source_v8 check WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." RESULT_VARIABLE ACTOR_RESULT)
if(NOT ACTOR_RESULT EQUAL 0)
 message(FATAL_ERROR "Derived actor mechanism differs from pinned original")
endif()
foreach(path "${LLAMA_ROOT}/ggml/include/ggml.h" "${STEADY_GENERATED_DIR}/state_response_generated.inc" "${CUDA_ROOT}/include/cuda_runtime_api.h" "${CUPTI_ROOT}/include/cupti.h")
 if(NOT EXISTS "${path}")
  message(FATAL_ERROR "Explicit dependency missing: ${path}")
 endif()
endforeach()
set(libraries)
foreach(component base cpu cuda)
 find_library(GGML_${component} NAMES ggml-${component} PATHS "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda" PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
 list(APPEND libraries "${GGML_${component}}")
endforeach()
find_library(CUDART_LIBRARY NAMES cudart PATHS "${CUDA_ROOT}/lib/x64" NO_DEFAULT_PATH REQUIRED)
find_library(CUPTI_LIBRARY NAMES cupti PATHS "${CUPTI_ROOT}/lib64" NO_DEFAULT_PATH REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/external_stage_probe.cpp" EXTERNAL_STAGE_V8_SOURCE_SHA256)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../v6/runtime_callbacks.hpp" EXTERNAL_STAGE_V8_CALLBACK_SHA256)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/actors.inc" EXTERNAL_STAGE_V8_ACTOR_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/external_stage_probe.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/actors.inc" "${CMAKE_CURRENT_SOURCE_DIR}/../v6/runtime_callbacks.hpp")
add_executable(dynamic-external-stage-v8 external_stage_probe.cpp)
target_compile_features(dynamic-external-stage-v8 PRIVATE cxx_std_17)
target_compile_definitions(dynamic-external-stage-v8 PRIVATE GGML_SHARED GGML_BACKEND_SHARED HIGHLEVEL_CUDA=1 EXTERNAL_STAGE_V8_SOURCE_SHA256="${EXTERNAL_STAGE_V8_SOURCE_SHA256}" EXTERNAL_STAGE_V8_CALLBACK_SHA256="${EXTERNAL_STAGE_V8_CALLBACK_SHA256}" EXTERNAL_STAGE_V8_ACTOR_SHA256="${EXTERNAL_STAGE_V8_ACTOR_SHA256}")
target_include_directories(dynamic-external-stage-v8 PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/ggml/src" "${LLAMA_ROOT}/vendor" "${CUDA_ROOT}/include" "${CUPTI_ROOT}/include" "${STEADY_GENERATED_DIR}")
target_link_libraries(dynamic-external-stage-v8 PRIVATE ${libraries} "${CUDART_LIBRARY}" "${CUPTI_LIBRARY}" bcrypt)
if(MSVC)
 target_compile_options(dynamic-external-stage-v8 PRIVATE /W4 /EHsc)
endif()
