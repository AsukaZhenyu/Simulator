cmake_minimum_required(VERSION 3.22)
project(dynamic_h2d_bare_v9 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows QPC required")
endif()
set(LLAMA_ROOT "" CACHE PATH "Pinned llama source")
set(GGML_BUILD_DIR "" CACHE PATH "Pinned existing GGML import libraries")
set(STEADY_GENERATED_DIR "" CACHE PATH "Existing generated Executor includes")
set(CUDA_ROOT "" CACHE PATH "Pinned toolkit")
set(CUPTI_ROOT "" CACHE PATH "Header-only dependency; not captured/linked")
set(libraries)
foreach(component base cpu cuda)
 find_library(GGML_${component} NAMES ggml-${component} PATHS "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda" PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
 list(APPEND libraries "${GGML_${component}}")
endforeach()
find_library(CUDART_LIBRARY NAMES cudart PATHS "${CUDA_ROOT}/lib/x64" NO_DEFAULT_PATH REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/h2d_bare.cpp" H2D_BARE_SOURCE_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/h2d_bare.cpp")
add_executable(dynamic-h2d-bare-v9 h2d_bare.cpp)
target_compile_features(dynamic-h2d-bare-v9 PRIVATE cxx_std_17)
target_compile_definitions(dynamic-h2d-bare-v9 PRIVATE GGML_SHARED GGML_BACKEND_SHARED HIGHLEVEL_CUDA=1 H2D_BARE_SOURCE_SHA256="${H2D_BARE_SOURCE_SHA256}")
target_include_directories(dynamic-h2d-bare-v9 PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/ggml/src" "${LLAMA_ROOT}/vendor" "${CUDA_ROOT}/include" "${CUPTI_ROOT}/include" "${STEADY_GENERATED_DIR}")
target_link_libraries(dynamic-h2d-bare-v9 PRIVATE ${libraries} "${CUDART_LIBRARY}" bcrypt)
if(MSVC)
 target_compile_options(dynamic-h2d-bare-v9 PRIVATE /W4 /EHsc)
endif()
