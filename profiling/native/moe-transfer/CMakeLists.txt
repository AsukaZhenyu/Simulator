cmake_minimum_required(VERSION 3.20)
project(moe_transfer_probe LANGUAGES CXX)
find_package(CUDAToolkit REQUIRED)
add_executable(moe-transfer-probe ../moe_transfer_probe.cpp)
target_compile_features(moe-transfer-probe PRIVATE cxx_std_17)
target_link_libraries(moe-transfer-probe PRIVATE CUDA::cudart)
if(MSVC)
    target_compile_options(moe-transfer-probe PRIVATE /W4 /EHsc)
endif()
