cmake_minimum_required(VERSION 3.20)
project(state_response_etw_offline_tools LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "These optional offline tools require Windows, MSVC and the Windows SDK")
endif()
set(NLOHMANN_JSON_INCLUDE_DIR "" CACHE PATH "Explicit directory containing nlohmann/json.hpp")
if(NOT EXISTS "${NLOHMANN_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Set NLOHMANN_JSON_INCLUDE_DIR explicitly; dependencies are never downloaded")
endif()
foreach(tool IN ITEMS etw_export etw_marker_export etw_interrupt_export)
  add_executable(${tool} "../${tool}.cpp")
  target_compile_features(${tool} PRIVATE cxx_std_17)
  target_compile_definitions(${tool} PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
  target_include_directories(${tool} PRIVATE "${NLOHMANN_JSON_INCLUDE_DIR}")
  target_compile_options(${tool} PRIVATE /utf-8 /EHsc)
  target_link_libraries(${tool} PRIVATE advapi32)
endforeach()
target_link_libraries(etw_export PRIVATE tdh)
