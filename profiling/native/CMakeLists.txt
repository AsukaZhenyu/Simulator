cmake_minimum_required(VERSION 3.20)
project(llm_infer_probes LANGUAGES CXX)

# Reuse an existing GGML build. These probes never rebuild or edit that source tree.
set(LLAMA_SOURCE_DIR "" CACHE PATH "Optional alias for a llama.cpp tree with build/")
set(GGML_SOURCE_DIR "" CACHE PATH "GGML source root (containing include/ggml.h), or its include directory")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML or llama.cpp build root")
option(MEASURE_ENABLE_CUDA "Build CUDA measurement support when dependencies exist" ON)
option(MEASURE_GGML_SHARED "Existing GGML libraries are shared libraries/import libraries" ON)
if(LLAMA_SOURCE_DIR)
    if(NOT GGML_SOURCE_DIR)
        set(GGML_SOURCE_DIR "${LLAMA_SOURCE_DIR}/ggml")
    endif()
    if(NOT GGML_BUILD_DIR)
        set(GGML_BUILD_DIR "${LLAMA_SOURCE_DIR}/build")
    endif()
endif()
find_path(MEASURE_GGML_INCLUDE ggml.h
    PATHS "${GGML_SOURCE_DIR}" "${GGML_SOURCE_DIR}/include" NO_DEFAULT_PATH)
if(NOT MEASURE_GGML_INCLUDE OR NOT GGML_BUILD_DIR)
    message(FATAL_ERROR "Set GGML_SOURCE_DIR (public headers) and GGML_BUILD_DIR (existing build)")
endif()
set(ggml_library_paths
    "${GGML_BUILD_DIR}" "${GGML_BUILD_DIR}/lib" "${GGML_BUILD_DIR}/lib64"
    "${GGML_BUILD_DIR}/src" "${GGML_BUILD_DIR}/ggml/src"
    "${GGML_BUILD_DIR}/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu"
    "${GGML_BUILD_DIR}/src/ggml-cuda" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda")
foreach(component IN ITEMS base cpu)
    find_library(MEASURE_GGML_${component}_LIBRARY NAMES ggml-${component}
        PATHS ${ggml_library_paths} PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH)
    if(NOT MEASURE_GGML_${component}_LIBRARY)
        message(FATAL_ERROR "Missing existing ggml-${component} library below GGML_BUILD_DIR")
    endif()
endforeach()
find_package(Threads REQUIRED)

add_executable(ggml-compute-probe compute_probe.cpp)
target_compile_features(ggml-compute-probe PRIVATE cxx_std_17)
target_include_directories(ggml-compute-probe PRIVATE "${MEASURE_GGML_INCLUDE}")
if(MEASURE_GGML_SHARED)
    target_compile_definitions(ggml-compute-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
endif()
target_link_libraries(ggml-compute-probe PRIVATE
    "${MEASURE_GGML_cpu_LIBRARY}" "${MEASURE_GGML_base_LIBRARY}" Threads::Threads)

# Macro profiles initially use CPU only. They do not borrow CUDA matmul prices.
add_executable(ggml-macro-probe macro_probe.cpp)
target_compile_features(ggml-macro-probe PRIVATE cxx_std_17)
target_include_directories(ggml-macro-probe PRIVATE "${MEASURE_GGML_INCLUDE}")
if(MEASURE_GGML_SHARED)
    target_compile_definitions(ggml-macro-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
endif()
target_link_libraries(ggml-macro-probe PRIVATE
    "${MEASURE_GGML_cpu_LIBRARY}" "${MEASURE_GGML_base_LIBRARY}" Threads::Threads)
# Complete dense decoder-block graphs with explicit materialized stage boundaries.
add_executable(ggml-dense-layer-probe dense_layer_probe.cpp)
target_compile_features(ggml-dense-layer-probe PRIVATE cxx_std_17)
target_include_directories(ggml-dense-layer-probe PRIVATE "${MEASURE_GGML_INCLUDE}")
if(MEASURE_GGML_SHARED)
    target_compile_definitions(ggml-dense-layer-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
endif()
target_link_libraries(ggml-dense-layer-probe PRIVATE
    "${MEASURE_GGML_cpu_LIBRARY}" "${MEASURE_GGML_base_LIBRARY}" Threads::Threads)
# Static CPU builds can need their original OpenMP dependency at the final link.
if(NOT MEASURE_GGML_SHARED)
    find_package(OpenMP QUIET COMPONENTS CXX)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(ggml-compute-probe PRIVATE OpenMP::OpenMP_CXX)
        target_link_libraries(ggml-macro-probe PRIVATE OpenMP::OpenMP_CXX)
        target_link_libraries(ggml-dense-layer-probe PRIVATE OpenMP::OpenMP_CXX)
    endif()
endif()
if(MSVC)
    target_compile_options(ggml-compute-probe PRIVATE /W4 /EHsc)
    target_compile_options(ggml-macro-probe PRIVATE /W4 /EHsc)
    target_compile_options(ggml-dense-layer-probe PRIVATE /W4 /EHsc)
endif()

# The runner supplies the matching runtime libraries; no machine-specific paths.
if(MEASURE_ENABLE_CUDA)
    find_package(CUDAToolkit QUIET)
    find_library(MEASURE_GGML_cuda_LIBRARY NAMES ggml-cuda
        PATHS ${ggml_library_paths} PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH)
    if(CUDAToolkit_FOUND AND MEASURE_GGML_cuda_LIBRARY AND EXISTS "${MEASURE_GGML_INCLUDE}/ggml-cuda.h")
        target_compile_definitions(ggml-compute-probe PRIVATE MEASURE_HAVE_CUDA=1)
        target_link_libraries(ggml-compute-probe PRIVATE "${MEASURE_GGML_cuda_LIBRARY}")
        # Independent complete CUDA decoder probe; CPU diagnostics are frozen.
        add_executable(ggml-dense-layer-cuda-probe dense_layer_cuda_probe.cpp)
        target_compile_features(ggml-dense-layer-cuda-probe PRIVATE cxx_std_17)
        target_include_directories(ggml-dense-layer-cuda-probe PRIVATE "${MEASURE_GGML_INCLUDE}")
        if(MEASURE_GGML_SHARED)
            target_compile_definitions(ggml-dense-layer-cuda-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
        endif()
        target_link_libraries(ggml-dense-layer-cuda-probe PRIVATE
            "${MEASURE_GGML_cuda_LIBRARY}" "${MEASURE_GGML_base_LIBRARY}" CUDA::cudart Threads::Threads)
        if(MSVC)
            target_compile_options(ggml-dense-layer-cuda-probe PRIVATE /W4 /EHsc)
        endif()
        # A separate binary adds synchronized memory checkpoints. Never price its timing.
        add_executable(ggml-dense-layer-cuda-memory-probe dense_layer_cuda_memory_probe.cpp)
        target_compile_features(ggml-dense-layer-cuda-memory-probe PRIVATE cxx_std_17)
        target_include_directories(ggml-dense-layer-cuda-memory-probe PRIVATE "${MEASURE_GGML_INCLUDE}")
        if(MEASURE_GGML_SHARED)
            target_compile_definitions(ggml-dense-layer-cuda-memory-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
        endif()
        target_link_libraries(ggml-dense-layer-cuda-memory-probe PRIVATE
            "${MEASURE_GGML_cuda_LIBRARY}" "${MEASURE_GGML_base_LIBRARY}" CUDA::cudart Threads::Threads)
        if(MSVC)
            target_compile_options(ggml-dense-layer-cuda-memory-probe PRIVATE /W4 /EHsc)
        endif()
        if(NOT MEASURE_GGML_SHARED)
            target_link_libraries(ggml-dense-layer-cuda-memory-probe PRIVATE CUDA::cublas CUDA::cublasLt)
        endif()
        if(NOT MEASURE_GGML_SHARED)
            target_link_libraries(ggml-dense-layer-cuda-probe PRIVATE CUDA::cublas CUDA::cublasLt)
            target_link_libraries(ggml-compute-probe PRIVATE CUDA::cudart CUDA::cublas CUDA::cublasLt)
        endif()
    else()
        message(STATUS "GGML CUDA dependencies unavailable: compute probe will support CPU only")
    endif()
    if(CUDAToolkit_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/transfer_probe.cpp")
        add_executable(cuda-transfer-probe transfer_probe.cpp)
        target_compile_features(cuda-transfer-probe PRIVATE cxx_std_17)
        target_link_libraries(cuda-transfer-probe PRIVATE CUDA::cudart)
    endif()
endif()
