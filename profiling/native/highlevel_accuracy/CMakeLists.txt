cmake_minimum_required(VERSION 3.22)
project(highlevel_accuracy LANGUAGES CXX)
option(HIGHLEVEL_ENABLE_CUDA "Enable real CUDA actions" ON)
set(LLAMA_ROOT "" CACHE PATH "Existing llama.cpp source root")
set(GGML_BUILD_DIR "" CACHE PATH "Existing llama.cpp build root")
if(NOT EXISTS "${LLAMA_ROOT}/ggml/include/ggml.h")
  message(FATAL_ERROR "Set LLAMA_ROOT to existing source containing ggml and vendor/nlohmann")
endif()
set(libpaths "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda")
foreach(component IN ITEMS base cpu)
  find_library(GGML_${component} NAMES ggml-${component} PATHS ${libpaths} PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
endforeach()
add_executable(highlevel-probe highlevel_probe.cpp)
target_compile_features(highlevel-probe PRIVATE cxx_std_17)
target_compile_definitions(highlevel-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
target_include_directories(highlevel-probe PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/vendor")
target_link_libraries(highlevel-probe PRIVATE "${GGML_cpu}" "${GGML_base}")
if(HIGHLEVEL_ENABLE_CUDA)
  find_library(GGML_cuda NAMES ggml-cuda PATHS ${libpaths} PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
  target_compile_definitions(highlevel-probe PRIVATE HIGHLEVEL_CUDA=1)
  target_link_libraries(highlevel-probe PRIVATE "${GGML_cuda}")
endif()
if(MSVC)
  target_compile_options(highlevel-probe PRIVATE /W4 /EHsc)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(record_libraries --link-library "${GGML_cpu}" --link-library "${GGML_base}")
if(HIGHLEVEL_ENABLE_CUDA)
  list(APPEND record_libraries --link-library "${GGML_cuda}")
endif()
add_custom_command(TARGET highlevel-probe POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build_record.py"
    --binary "$<TARGET_FILE:highlevel-probe>" --source-dir "${CMAKE_CURRENT_SOURCE_DIR}"
    --compiler "${CMAKE_CXX_COMPILER}" --flags "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE} ${CMAKE_EXE_LINKER_FLAGS_RELEASE}"
    ${record_libraries}
  VERBATIM)
