cmake_minimum_required(VERSION 3.24)
project(overlap_accuracy_probe LANGUAGES CXX)
set(GGML_SOURCE_DIR "" CACHE PATH "Existing GGML source root")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML shared-library build root")
find_package(CUDAToolkit REQUIRED)
find_path(PROBE_INCLUDE ggml.h PATHS "${GGML_SOURCE_DIR}/include" NO_DEFAULT_PATH REQUIRED)
foreach(part base cpu cuda)
  find_library(PROBE_${part} NAMES ggml-${part} PATHS "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda" PATH_SUFFIXES Release REQUIRED NO_DEFAULT_PATH)
endforeach()
set(CONTEXT "${CMAKE_CURRENT_SOURCE_DIR}/../execution_transforms_dual/compute_context.cpp")
set(MATCHED "${CMAKE_CURRENT_SOURCE_DIR}/../matched_mechanisms/compute.cpp")
file(SHA256 "${CONTEXT}" CONTEXT_SHA)
file(SHA256 "${MATCHED}" MATCHED_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/compute.cpp" PROBE_SHA)
if(NOT CONTEXT_SHA STREQUAL "37ed5525074d2653050daa433a7b938a6f7f751a1afdeaac17a8c21cc2173cd4" OR NOT MATCHED_SHA STREQUAL "2556bed47f9f45ffef7884f5f58b452ec69a5c036b98a1e137afc9c55085c327")
  message(FATAL_ERROR "Immutable helper source changed")
endif()
file(READ "${CONTEXT}" original)
string(REPLACE "int main(int argc,char**argv)" "static int overlap_context_legacy_main(int argc,char**argv)" generated "${original}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/context_generated.inc" "${generated}")
add_executable(overlap-accuracy-probe compute.cpp)
target_compile_features(overlap-accuracy-probe PRIVATE cxx_std_17)
target_include_directories(overlap-accuracy-probe PRIVATE "${PROBE_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../execution_transforms_dual")
target_compile_definitions(overlap-accuracy-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED MATCHED_HAVE_CUDA OVERLAP_PROBE_SHA="${PROBE_SHA}" OVERLAP_CONTEXT_SHA="${CONTEXT_SHA}" OVERLAP_MATCHED_SHA="${MATCHED_SHA}")
target_link_libraries(overlap-accuracy-probe PRIVATE "${PROBE_base}" "${PROBE_cpu}" "${PROBE_cuda}" CUDA::cudart)
