cmake_minimum_required(VERSION 3.24)
project(workload_accuracy LANGUAGES CXX)
set(GGML_SOURCE_DIR "" CACHE PATH "Existing GGML source root")
set(GGML_BUILD_DIR "" CACHE PATH "Existing llama.cpp/GGML build root")
set(NLOHMANN_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
option(WORKLOAD_CUDA "Build CUDA support" ON)
find_path(PROBE_INCLUDE ggml.h PATHS "${GGML_SOURCE_DIR}/include" NO_DEFAULT_PATH REQUIRED)
find_path(JSON_INCLUDE nlohmann/json.hpp PATHS "${NLOHMANN_INCLUDE_DIR}" "${GGML_SOURCE_DIR}/../vendor" NO_DEFAULT_PATH REQUIRED)
set(libpaths "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda")
foreach(part base cpu)
  find_library(PROBE_${part} NAMES ggml-${part} PATHS ${libpaths} PATH_SUFFIXES Release REQUIRED NO_DEFAULT_PATH)
endforeach()
find_package(Threads REQUIRED)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_executable(ggml-workload-accuracy-probe workload_probe.cpp)
target_compile_features(ggml-workload-accuracy-probe PRIVATE cxx_std_17)
target_include_directories(ggml-workload-accuracy-probe PRIVATE "${PROBE_INCLUDE}" "${JSON_INCLUDE}")
target_compile_definitions(ggml-workload-accuracy-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
target_link_libraries(ggml-workload-accuracy-probe PRIVATE "${PROBE_base}" "${PROBE_cpu}" Threads::Threads)
if(WORKLOAD_CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_library(PROBE_cuda NAMES ggml-cuda PATHS ${libpaths} PATH_SUFFIXES Release REQUIRED NO_DEFAULT_PATH)
  target_compile_definitions(ggml-workload-accuracy-probe PRIVATE WORKLOAD_HAVE_CUDA)
  target_link_libraries(ggml-workload-accuracy-probe PRIVATE "${PROBE_cuda}" CUDA::cudart)
endif()
if(MSVC)
  target_compile_options(ggml-workload-accuracy-probe PRIVATE /W4 /EHsc /fp:precise)
endif()
add_custom_command(TARGET ggml-workload-accuracy-probe POST_BUILD
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/record_build.py"
    --binary "$<TARGET_FILE:ggml-workload-accuracy-probe>" --source-dir "${CMAKE_CURRENT_SOURCE_DIR}"
    --ggml-source "${GGML_SOURCE_DIR}" --ggml-build "${GGML_BUILD_DIR}"
    --compiler "${CMAKE_CXX_COMPILER}" --compiler-version "${CMAKE_CXX_COMPILER_VERSION}"
  VERBATIM)
