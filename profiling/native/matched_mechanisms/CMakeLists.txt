cmake_minimum_required(VERSION 3.24)
project(matched_mechanism_probes LANGUAGES CXX)
set(GGML_SOURCE_DIR "" CACHE PATH "Existing GGML source root")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML build root")
option(MATCHED_ENABLE_CUDA "Enable CUDA; never fallback for a GPU request" ON)
find_path(MATCHED_INCLUDE ggml.h PATHS "${GGML_SOURCE_DIR}/include" NO_DEFAULT_PATH REQUIRED)
set(libpaths "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda"
    "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/src"
    "${GGML_BUILD_DIR}/src/ggml-cuda" "${GGML_BUILD_DIR}/src/ggml-cpu")
foreach(part base cpu)
  find_library(MATCHED_${part} NAMES ggml-${part} PATHS ${libpaths} PATH_SUFFIXES Release REQUIRED NO_DEFAULT_PATH)
endforeach()
add_executable(matched-compute-probe compute.cpp)
target_compile_features(matched-compute-probe PRIVATE cxx_std_17)
target_include_directories(matched-compute-probe PRIVATE "${MATCHED_INCLUDE}")
target_compile_definitions(matched-compute-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
target_link_libraries(matched-compute-probe PRIVATE "${MATCHED_base}" "${MATCHED_cpu}")
if(MATCHED_ENABLE_CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_library(MATCHED_cuda NAMES ggml-cuda PATHS ${libpaths} PATH_SUFFIXES Release REQUIRED NO_DEFAULT_PATH)
  target_compile_definitions(matched-compute-probe PRIVATE MATCHED_HAVE_CUDA)
  target_link_libraries(matched-compute-probe PRIVATE "${MATCHED_cuda}" CUDA::cudart)
  enable_language(CUDA)
  add_executable(matched-transfer-probe transfer.cu)
  set_target_properties(matched-transfer-probe PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES native)
  target_link_libraries(matched-transfer-probe PRIVATE CUDA::cudart)
endif()
