cmake_minimum_required(VERSION 3.20)
project(moe_probe LANGUAGES CXX)
set(GGML_SOURCE_DIR "" CACHE PATH "GGML source tree")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML build")
find_path(MOE_INCLUDE ggml.h PATHS "${GGML_SOURCE_DIR}/include" NO_DEFAULT_PATH)
set(moe_paths "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/src" "${GGML_BUILD_DIR}/src/ggml-cpu")
foreach(component IN ITEMS base cpu)
 find_library(MOE_${component}_LIBRARY NAMES ggml-${component} PATHS ${moe_paths} PATH_SUFFIXES Release NO_DEFAULT_PATH)
 if(NOT MOE_${component}_LIBRARY)
  message(FATAL_ERROR "Missing ggml-${component}")
 endif()
endforeach()
if(NOT MOE_INCLUDE)
 message(FATAL_ERROR "GGML public headers unavailable")
endif()
add_executable(ggml-moe-probe ../moe_probe.cpp)
target_compile_features(ggml-moe-probe PRIVATE cxx_std_17)
target_include_directories(ggml-moe-probe PRIVATE "${MOE_INCLUDE}")
target_compile_definitions(ggml-moe-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED)
target_link_libraries(ggml-moe-probe PRIVATE "${MOE_cpu_LIBRARY}" "${MOE_base_LIBRARY}")
if(MSVC)
 target_compile_options(ggml-moe-probe PRIVATE /W4 /EHsc)
endif()
