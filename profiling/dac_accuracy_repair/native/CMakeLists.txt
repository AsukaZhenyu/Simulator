cmake_minimum_required(VERSION 3.22)
project(dac_accuracy_repair LANGUAGES CXX)
set(LLAMA_ROOT "" CACHE PATH "Explicit existing llama.cpp source")
set(GGML_BUILD_DIR "" CACHE PATH "Existing GGML import libraries")
set(STEADY_GENERATED_DIR "" CACHE PATH "Pinned pre-existing generated Executor files")
set(CUDA_ROOT "" CACHE PATH "Installed CUDA toolkit")
foreach(path "${LLAMA_ROOT}/ggml/include/ggml.h" "${STEADY_GENERATED_DIR}/state_response_generated.inc" "${CUDA_ROOT}/include/cuda_runtime_api.h")
 if(NOT EXISTS "${path}")
  message(FATAL_ERROR "Explicit file missing: ${path}")
 endif()
endforeach()
set(libraries)
foreach(component base cpu cuda)
 find_library(GGML_${component} NAMES ggml-${component} PATHS "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda" PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
 list(APPEND libraries "${GGML_${component}}")
endforeach()
find_library(CUDART_LIBRARY NAMES cudart PATHS "${CUDA_ROOT}/lib/x64" NO_DEFAULT_PATH REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/repair_probe.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/repair_probe.cpp" REPAIR_SOURCE_SHA256)
add_executable(accuracy-repair-probe repair_probe.cpp)
target_compile_features(accuracy-repair-probe PRIVATE cxx_std_17)
target_compile_definitions(accuracy-repair-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED HIGHLEVEL_CUDA=1 REPAIR_SOURCE_SHA256="${REPAIR_SOURCE_SHA256}")
target_include_directories(accuracy-repair-probe PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/ggml/src" "${LLAMA_ROOT}/vendor" "${CUDA_ROOT}/include" "${STEADY_GENERATED_DIR}")
target_link_libraries(accuracy-repair-probe PRIVATE ${libraries} "${CUDART_LIBRARY}" bcrypt)
if(MSVC)
 target_compile_options(accuracy-repair-probe PRIVATE /W4 /EHsc)
endif()
