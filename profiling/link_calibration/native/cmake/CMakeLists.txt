cmake_minimum_required(VERSION 3.22)
project(link_copy_calibration LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "This bounded probe requires Windows QPC and explicit CUDA0")
endif()
set(LLAMA_ROOT "" CACHE PATH "Matching retained llama.cpp source")
set(GGML_BUILD_DIR "" CACHE PATH "Matching retained GGML import libraries")
set(STEADY_GENERATED_DIR "" CACHE PATH "Bound unchanged generated Executor")
set(CUDA_ROOT "" CACHE PATH "Installed CUDA toolkit")
foreach(path "${LLAMA_ROOT}/ggml/include/ggml.h" "${STEADY_GENERATED_DIR}/state_response_generated.inc" "${CUDA_ROOT}/include/cuda_runtime_api.h")
 if(NOT EXISTS "${path}")
  message(FATAL_ERROR "Explicit matching dependency missing: ${path}")
 endif()
endforeach()
set(libraries)
foreach(component base cpu cuda)
 find_library(GGML_${component} NAMES ggml-${component} PATHS "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
 list(APPEND libraries "${GGML_${component}}")
endforeach()
find_library(CUDART_LIBRARY NAMES cudart PATHS "${CUDA_ROOT}/lib/x64" NO_DEFAULT_PATH REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../link_probe.cpp" LINK_COPY_SOURCE_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../link_probe.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/../pcie_nvml.hpp")
add_executable(link-copy-probe ../link_probe.cpp)
target_compile_features(link-copy-probe PRIVATE cxx_std_17)
target_compile_definitions(link-copy-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED HIGHLEVEL_CUDA=1 LINK_COPY_SOURCE_SHA256="${LINK_COPY_SOURCE_SHA256}")
target_include_directories(link-copy-probe PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/ggml/src" "${LLAMA_ROOT}/vendor" "${CUDA_ROOT}/include" "${STEADY_GENERATED_DIR}")
target_link_libraries(link-copy-probe PRIVATE ${libraries} "${CUDART_LIBRARY}" bcrypt)
if(MSVC)
 target_compile_options(link-copy-probe PRIVATE /O2 /EHsc /utf-8 /bigobj)
endif()
