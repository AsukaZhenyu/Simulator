cmake_minimum_required(VERSION 3.22)
project(moe_expert_policy LANGUAGES CXX)
set(LLAMA_ROOT "" CACHE PATH "Explicit existing llama.cpp source")
set(GGML_BUILD_DIR "" CACHE PATH "Explicit existing llama.cpp build")
if(NOT EXISTS "${LLAMA_ROOT}/ggml/include/ggml.h")
  message(FATAL_ERROR "Explicit matching LLAMA_ROOT required")
endif()
set(libpaths "${GGML_BUILD_DIR}/ggml/src" "${GGML_BUILD_DIR}/ggml/src/ggml-cpu" "${GGML_BUILD_DIR}/ggml/src/ggml-cuda")
foreach(component base cpu cuda)
  find_library(GGML_${component} NAMES ggml-${component} PATHS ${libpaths} PATH_SUFFIXES Release RelWithDebInfo NO_DEFAULT_PATH REQUIRED)
  list(APPEND libraries "${GGML_${component}}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp" PROBE_SOURCE_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp")
add_executable(moe-expert-policy-probe probe.cpp)
target_compile_features(moe-expert-policy-probe PRIVATE cxx_std_17)
target_compile_definitions(moe-expert-policy-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED PROBE_SOURCE_SHA256="${PROBE_SOURCE_SHA256}")
target_include_directories(moe-expert-policy-probe PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/vendor")
target_link_libraries(moe-expert-policy-probe PRIVATE ${libraries})
if(MSVC)
  target_compile_options(moe-expert-policy-probe PRIVATE /W4 /EHsc /utf-8)
endif()
