cmake_minimum_required(VERSION 3.22)
project(real_moe_validation LANGUAGES CXX)
set(LLAMA_ROOT "" CACHE PATH "Explicit matching llama.cpp source")
set(LLAMA_BUILD "" CACHE PATH "Existing matching llama.cpp build")
foreach(path "${LLAMA_ROOT}/include/llama.h" "${LLAMA_ROOT}/vendor/nlohmann/json.hpp" "${LLAMA_ROOT}/src/llama-model.h")
 if(NOT EXISTS "${path}")
  message(FATAL_ERROR "Explicit matching input missing: ${path}")
 endif()
endforeach()
find_library(LLAMA_LIBRARY NAMES llama PATHS "${LLAMA_BUILD}/src/Release" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_BASE_LIBRARY NAMES ggml-base PATHS "${LLAMA_BUILD}/ggml/src/Release" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_LIBRARY NAMES ggml PATHS "${LLAMA_BUILD}/ggml/src/Release" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_CPU_LIBRARY NAMES ggml-cpu PATHS "${LLAMA_BUILD}/ggml/src/Release" NO_DEFAULT_PATH REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/real_moe_probe.cpp" REAL_MOE_SOURCE_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/real_moe_probe.cpp")
add_executable(real-moe-probe real_moe_probe.cpp)
target_compile_features(real-moe-probe PRIVATE cxx_std_17)
target_compile_definitions(real-moe-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED REAL_MOE_SOURCE_SHA256="${REAL_MOE_SOURCE_SHA256}")
target_include_directories(real-moe-probe PRIVATE "${LLAMA_ROOT}/include" "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/src" "${LLAMA_ROOT}/vendor")
target_link_libraries(real-moe-probe PRIVATE "${LLAMA_LIBRARY}" "${GGML_LIBRARY}" "${GGML_BASE_LIBRARY}" "${GGML_CPU_LIBRARY}")
if(MSVC)
 target_compile_options(real-moe-probe PRIVATE /W4 /EHsc /utf-8 /bigobj)
endif()
