cmake_minimum_required(VERSION 3.22)
project(real_moe_quantized_subgraphs LANGUAGES CXX)
set(LLAMA_ROOT "" CACHE PATH "Explicit matching llama.cpp source")
set(LLAMA_BUILD "" CACHE PATH "Explicit matching llama.cpp build")
foreach(name ggml ggml-base ggml-cpu)
 string(REPLACE "-" "_" key "${name}")
 find_library(${key}_LIBRARY NAMES ${name} PATHS "${LLAMA_BUILD}/ggml/src/Release" NO_DEFAULT_PATH REQUIRED)
 list(APPEND PROFILE_LIBRARIES "${${key}_LIBRARY}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/quantized_probe.cpp" QUANTIZED_SOURCE_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/quantized_probe.cpp")
add_executable(real-moe-quantized-probe quantized_probe.cpp)
target_compile_features(real-moe-quantized-probe PRIVATE cxx_std_17)
target_compile_definitions(real-moe-quantized-probe PRIVATE GGML_SHARED GGML_BACKEND_SHARED QUANTIZED_SOURCE_SHA256="${QUANTIZED_SOURCE_SHA256}")
target_include_directories(real-moe-quantized-probe PRIVATE "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/vendor")
target_link_libraries(real-moe-quantized-probe PRIVATE ${PROFILE_LIBRARIES})
if(MSVC)
 target_compile_options(real-moe-quantized-probe PRIVATE /W4 /EHsc /utf-8 /bigobj)
endif()
